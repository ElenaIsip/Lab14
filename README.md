# Алгоритм и блок-схема
Начало.

Установить русскоязычную локаль для корректного отображения текста в консоли.

Запросить у пользователя количество элементов массива n.

Проверка: если ввод некорректен (введено не число или n <= 0), вывести ошибку и завершить программу.

Выделить память под массив из n целых чисел с помощью malloc.

Проверка: если память не выделилась (указатель arr == NULL), вывести ошибку и завершить программу.

Запустить цикл от i = 0 до n - 1:

Запросить ввод целого числа.

Считать число и сохранить в arr[i].

Если ввод некорректен, освободить память, вывести ошибку и завершить программу.

Инициализировать переменную last_index = -1 (означает, что искомый элемент пока не найден).

Запустить цикл от i = 0 до n - 1:

Проверка: если arr[i] < 0 (число отрицательное) И arr[i] % 3 == 0 (делится на 3 без остатка):

Присвоить last_index = i (запоминаем индекс).

Проверка: если last_index != -1 (элемент был найден):

Вывести номер элемента (last_index + 1) и его значение arr[last_index].

Иначе:

Вывести сообщение, что таких элементов нет.

Освободить выделенную память с помощью free(arr).

 # Реализация программы

     #define _CRT_SECURE_NO_WARNINGS
     #include <stdio.h>
    #include <stdlib.h> // Необходима для malloc() и free()
    #include <locale.h>

     int main(void) {
    setlocale(LC_ALL, "Russian");

    int n;

    // 1. Ввод количества элементов массива
    printf("Введите количество элементов массива (n): ");
    if (scanf("%d", &n) != 1 || n <= 0) {
        printf("Ошибка: n должно быть положительным целым числом.\n");
        return 1;
    }

    // 2. Выделение памяти под массив размера n
    int* arr = (int*)malloc(n * sizeof(int));

    // Проверка, что память успешно выделилась
    if (arr == NULL) {
        printf("Ошибка: не удалось выделить память под массив.\n");
        return 1;
    }

    // 3. Ввод элементов массива с клавиатуры
    printf("Введите %d целых чисел (положительных и отрицательных):\n", n);
    for (int i = 0; i < n; i++) {
        printf("arr[%d] = ", i + 1);
        if (scanf("%d", &arr[i]) != 1) {
            printf("Ошибка ввода числа.\n");
            free(arr); // Освобождаем память перед выходом, чтобы не было утечек
            return 1;
        }
    }

    // 4. Поиск номера последнего отрицательного элемента, делящегося на 3
    int last_index = -1; // -1 означает, что элемент не найден

    for (int i = 0; i < n; i++) {
        // Условие: число отрицательное И делится на 3 без остатка
        if (arr[i] < 0 && arr[i] % 3 == 0) {
            last_index = i;
        }
    }

    // 5. Вывод результата
    if (last_index != -1) {
        printf("\nНомер последнего отрицательного элемента, делящегося на 3 без остатка: %d\n", last_index + 1);
        printf("Значение этого элемента: %d\n", arr[last_index]);
    }
    else {
        printf("\nВ массиве нет отрицательных элементов, делящихся на 3 без остатка.\n");
    }

    // 6. Освобождение выделенной памяти
    free(arr);

    return 0;
    }
